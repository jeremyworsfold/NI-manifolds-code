% Shared colours, typography and plot styles for the manuscript figures.
\providecommand{\figtick}{\footnotesize}
\providecommand{\figlabel}{\small}
\providecommand{\figheading}{\normalsize}

\definecolor{fastgrey}{HTML}{6F6F6F}
\definecolor{leading}{HTML}{0072B2}
\definecolor{firstorder}{HTML}{A64D79}
\definecolor{stateorange}{HTML}{E69F00}

\tikzset{
    sublabel/.style = {
        font = \figlabel,
        inner sep = 0pt,
    },
    keyline/.style = {
        line width = 0.8pt,
    },
    keytext/.style = {
        font = \figtick,
        text = black!75,
        anchor = west,
        inner sep = 2pt,
    },
    manifold/.style = {
        draw = leading,
        line width = 1.4pt,
        line cap = round,
    },
    fibre/.style = {
        draw = black!18,
        line width = 0.4pt,
    },
    driftbox/.style = {
        draw = black!30,
        line width = 0.5pt,
        rounded corners = 3pt,
    },
    noisebox/.style = {
        draw = leading!70,
        line width = 0.5pt,
        rounded corners = 3pt,
    },
    slowarrow/.style = {
        -{Latex[length = 5pt, width = 4.5pt]},
        draw = leading,
        line width = 1.3pt,
        line cap = round,
    },
    fastarrow/.style = {
        ->,
        draw = fastgrey,
        line width = 0.9pt,
        line cap = round,
        shorten < = 2.5pt,
        shorten > = 2.5pt,
    },
    noisyarrow/.style = {
        fastarrow,
        decorate,
        decoration = {
            snake,
            segment length = 6pt,
            amplitude = 0.8pt,
            pre length = 2pt,
            post length = 3pt,
        },
    },
    state/.style = {
        circle,
        fill = stateorange,
        draw = white,
        line width = 0.6pt,
        minimum size = 0.2cm,
        inner sep = 0pt,
    },
    zero/.style = {
        state,
        fill = leading,
    },
    fixedpt/.style = {
        circle,
        fill = black,
        inner sep = 0pt,
        minimum size = 3.5pt,
    },
}

    },
